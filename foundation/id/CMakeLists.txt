cmake_minimum_required(VERSION 3.20)
project(utility_id_tests C)
include(CTest)
add_executable(utility_id_tests src/utility_id.c tests/test_utility_id.c)
target_include_directories(utility_id_tests PRIVATE include tests)
target_compile_features(utility_id_tests PRIVATE c_std_11)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(utility_id_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME utility_id_unit COMMAND utility_id_tests)
