cmake_minimum_required(VERSION 3.20)
project(utility_thread_pool_tests C)
include(CTest)
find_package(Threads REQUIRED)
add_executable(utility_thread_pool_tests
    src/utility_thread_pool.c tests/test_utility_thread_pool.c)
target_include_directories(utility_thread_pool_tests PRIVATE include tests)
target_compile_features(utility_thread_pool_tests PRIVATE c_std_11)
target_link_libraries(utility_thread_pool_tests PRIVATE Threads::Threads)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(utility_thread_pool_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME utility_thread_pool_unit COMMAND utility_thread_pool_tests)
set_tests_properties(utility_thread_pool_unit PROPERTIES TIMEOUT 10)
